// Pune Ganpati LIVE: Firestore rules (hardened v2)
// Paste the WHOLE file in Firebase Console > Firestore Database > Rules > Publish.
// Never edit by guessing: every limit below is on purpose (see docs/OPERATIONS.md).

service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- helpers ----------
    // Only the owner Google account is admin. email_verified must be true.
    // To add a second admin later, use:  request.auth.token.email in ['a@gmail.com', 'b@gmail.com']
    // Add ONLY an address you fully control.
    function isAdmin() {
      return request.auth != null
          && request.auth.token.email == 'shubhamshri.1110@gmail.com'
          && request.auth.token.email_verified == true;
    }
    // string field d[k] exists-or-defaults-to-empty, and its length is within [min, max]
    function str(d, k, min, max) {
      return d.get(k, '') is string
          && d.get(k, '').size() >= min
          && d.get(k, '').size() <= max;
    }
    // optional YYYY-MM-DD (empty is allowed)
    function dateOrEmpty(d, k) {
      return d.get(k, '') is string
          && (d.get(k, '') == '' || d.get(k, '').matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$'));
    }
    // a listing request (public form, organiser portal). Every text field has a size cap.
    function okSub(d) {
      return str(d, 'title', 3, 150)
          && str(d, 'city', 1, 40)
          && str(d, 'category', 1, 40)
          && str(d, 'venue', 1, 200)
          && str(d, 'area', 0, 100)
          && str(d, 'organizer', 0, 120)
          && str(d, 'contact', 7, 60)
          && str(d, 'start_time', 0, 30)
          && str(d, 'passes_text', 0, 300)
          && str(d, 'image_url', 0, 500)
          && d.get('image_url', '').matches('^(https?://.*)?$')
          && str(d, 'description', 0, 3000)
          && str(d, 'image_data', 0, 600000)
          && d.get('start_date', '') is string
          && d.get('start_date', '').matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
          && dateOrEmpty(d, 'end_date');
    }

    // ---------- Ganpati guide: anonymous comments ----------
    match /comments/{commentId} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['mandalId', 'mandalName', 'text', 'displayName', 'createdAt', 'language', 'status'])
          && request.resource.data.keys().hasAll(['mandalId', 'mandalName', 'text', 'createdAt', 'language', 'status'])
          && request.resource.data.mandalId is string
          && request.resource.data.mandalId.matches('^[a-z0-9-]{1,60}$')
          && str(request.resource.data, 'mandalName', 1, 120)
          && str(request.resource.data, 'text', 1, 4000)
          && str(request.resource.data, 'displayName', 0, 60)
          && request.resource.data.createdAt == request.time
          && request.resource.data.language in ['en', 'mr', 'hi']
          && request.resource.data.status == 'approved';
      allow update: if false;
      allow delete: if isAdmin();          // admin can remove spam from the Moderation tab
    }

    // ---------- Ganpati guide: anonymous crowd reports ----------
    // crowdLevel / queueTime are fixed lists on purpose: free text here was an XSS hole.
    match /crowdReports/{reportId} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['mandalId', 'mandalName', 'crowdLevel', 'queueTime', 'createdAt'])
          && request.resource.data.mandalId is string
          && request.resource.data.mandalId.matches('^[a-z0-9-]{1,60}$')
          && str(request.resource.data, 'mandalName', 1, 120)
          && request.resource.data.crowdLevel in ['Comfortable', 'Moderate', 'Busy', 'Very Busy', 'Extremely Busy']
          && request.resource.data.queueTime in ['<10 min', '10–30 min', '30–60 min', '60+ min', '10-30 min', '30-60 min']
          && request.resource.data.createdAt == request.time;
      allow update: if false;
      allow delete: if isAdmin();
    }

    // ---------- Events (published ones are public) ----------
    match /events/{id} {
      allow read: if resource.data.published == true || isAdmin();
      allow create, delete: if isAdmin();
      // admin can change anything; the organiser who owns an event may ONLY take it down (published = false)
      allow update: if isAdmin()
        || (request.auth != null
            && resource.data.get('ownerUid', '') == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['published', 'updatedAt'])
            && request.resource.data.updatedAt == request.time
            && request.resource.data.published == false);
    }

    match /products/{id} {
      allow read: if resource.data.published == true || isAdmin();
      allow write: if isAdmin();
    }

    // site settings: public read (WhatsApp number, contact email, flags), admin write
    match /settings/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ---------- Listing requests (public form + organiser portal) ----------
    // Contact phone/email live ONLY here. Nobody except the admin and the owner can read them.
    match /submissions/{id} {
      // Only auto-generated ids (20 letters/digits). This blocks attackers from choosing weird ids.
      allow create: if id.matches('^[A-Za-z0-9]{20}$')
          && request.resource.data.keys().hasOnly(['title', 'city', 'category', 'venue', 'area', 'start_date', 'end_date', 'start_time', 'passes_text', 'image_url', 'image_data', 'description', 'organizer', 'contact', 'status', 'createdAt', 'ownerUid', 'ownerEmail'])
          && request.resource.data.keys().hasAll(['title', 'city', 'category', 'venue', 'start_date', 'contact', 'status', 'createdAt'])
          && request.resource.data.status == 'pending'
          && request.resource.data.createdAt == request.time
          && okSub(request.resource.data)
          // ownerUid and ownerEmail go together, and only for the signed-in user themself
          && (
               (!('ownerUid' in request.resource.data) && !('ownerEmail' in request.resource.data))
               || (request.auth != null
                   && request.resource.data.ownerUid == request.auth.uid
                   && request.resource.data.ownerEmail == request.auth.token.email)
             );

      // admin reads everything; an organiser reads only their own
      allow read: if isAdmin()
          || (request.auth != null && resource.data.get('ownerUid', '') == request.auth.uid);

      // organiser: edit content (goes back to pending review) or mark as withdrawn; never touch owner/event/admin fields
      allow update: if isAdmin()
        || (request.auth != null
            && resource.data.get('ownerUid', '') == request.auth.uid
            && request.resource.data.ownerUid == resource.data.ownerUid
            && ((request.resource.data.status == 'pending'
                 && okSub(request.resource.data)
                 && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['title', 'city', 'category', 'venue', 'area', 'start_date', 'end_date', 'start_time', 'passes_text', 'image_url', 'image_data', 'description', 'organizer', 'contact', 'status', 'updatedAt'])
                 && request.resource.data.updatedAt == request.time)
                || (request.resource.data.status == 'withdrawn'
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'updatedAt'])
                    && request.resource.data.updatedAt == request.time)));

      // organiser can delete only requests that are not tied to a live event
      allow delete: if isAdmin()
        || (request.auth != null
            && resource.data.get('ownerUid', '') == request.auth.uid
            && (resource.data.status in ['rejected', 'withdrawn']
                || (resource.data.status == 'pending' && !('eventId' in resource.data))));
    }

    // ---------- Everything else is locked ----------
    // (the GitHub Action in Phase 2 uses firebase-admin, which bypasses these rules)
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
